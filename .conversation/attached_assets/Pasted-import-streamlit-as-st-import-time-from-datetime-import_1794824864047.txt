import streamlit as st
import time
from datetime import datetime

# Set tetapan halaman mesra peranti mudah alih
st.set_page_config(page_title="MediTrack AI", page_icon="💊", layout="centered")

# --- DATA SIMULASI (PANGKALAN DATA) ---
if 'schedule' not in st.session_state:
    st.session_state.schedule = [
        {"id": 1, "name": "Metformin (Kencing Manis)", "time": "08:00 AM", "dose": "1 Biji", "status": "Belum Ambil", "color": "Putih Bulat"},
        {"id": 2, "name": "Amlodipine (Darah Tinggi)", "time": "08:00 PM", "dose": "1/2 Biji", "status": "Belum Ambil", "color": "Kuning Segi Empat"},
        {"id": 3, "name": "Paracetamol (Tahan Sakit)", "time": "12:00 PM", "dose": "2 Biji", "status": "Diambil", "color": "Putih Lonjong"}
    ]

# --- SENARIO NOTIFIKASI SUARA (SIMULASI PENGGERA) ---
current_time_now = datetime.now().strftime("%I:%M %p")

# --- ANTARAMUKA PENGGUNA (UI) APP ---
st.title("💊 MediTrack AI")
st.subheader("Hab Peringatan & Imbasan Ubat Pintar")
st.write("---")

# Bahagian 1: Profil Pesakit & Status Harian
st.sidebar.header("👤 Profil Pengguna")
st.sidebar.write("**Nama:** Pak Cik Ahmad")
st.sidebar.write("**Umur:** 67 Tahun")
st.sidebar.write("**Tahap Kepatuhan Harian:** 75%")

# Bahagian 2: Senarai Jadual Ubat (Dashboard)
st.header("📅 Jadual Ubat Hari Ini")
for item in st.session_state.schedule:
    col1, col2, col3 = st.columns([2, 1, 1])
    with col1:
        st.write(f"**{item['name']}**")
        st.caption(f"Masa: {item['time']} | Dos: {item['dose']}")
    with col2:
        if item['status'] == "Diambil":
            st.success(item['status'])
        else:
            st.warning(item['status'])
    with col3:
        if item['status'] != "Diambil":
            if st.button("Sahkan Ubat", key=f"btn_{item['id']}"):
                st.session_state.selected_item = item
                st.rerun()

st.write("---")

# Bahagian 3: Mod Imbasan AI (Kamera)
if 'selected_item' in st.session_state:
    item = st.session_state.selected_item
    st.header("📸 Kamera Imbasan AI Visi Komputer")
    st.info(f"Sila halakan kamera ke arah ubat: **{item['name']}** ({item['color']})")
    
    # Menggunakan komponen kamera bawaan Streamlit
    img_file = st.camera_input("Ambil gambar ubat untuk pengesahan AI")
    
    if img_file:
        with st.spinner("AI sedang menganalisis bentuk, saiz, dan warna ubat..."):
            time.sleep(2) # Simulasi pemprosesan API AI
            
            # Simulasi logik pengecaman AI berdasarkan data ubat
            st.subheader("🔮 Hasil Analisis API Visi Komputer:")
            st.success("✅ PENGESAHAN: UBAT ADALAH BETUL!")
            st.write(f"**Ubat Dikesan:** {item['name']}")
            st.write(f"**Padanan Ciri Visual:** {item['color']} (Ketepatan 98.4%)")
            st.write(f"**Arahan Dos:** Sila ambil **{item['dose']}** sekarang dengan air suam.")
            
            # Kemas kini status dalam pangkalan data simulasi
            for s in st.session_state.schedule:
                if s['id'] == item['id']:
                    s['status'] = "Diambil"
            
            # Butang selesai
            if st.button("Selesai & Log Pengambilan"):
                del st.session_state.selected_item
                st.rerun()
else:
    st.info("💡 Tip Harian: Klik butang 'Sahkan Ubat' pada mana-mana jadual di atas untuk mengaktifkan kamera pengesahan AI.")
